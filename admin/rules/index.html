rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // The HV Test admin: the Firebase account ID shown on hv-tests/admin after signing in
    function isAdmin() {
      return request.auth != null && request.auth.uid == 'vbqJe9D5IAYSBytSzM850motekm2';
    }
    // ---------- Team: people the owner gave admin access to, by Google email (/admin/team) ----------
    // team/{email}.p holds a level per area: 0 no access, 1 view, 2 edit. The owner (isAdmin) can do everything.
    function myEmail() { return request.auth.token.email.lower(); }
    function isMember() {
      return request.auth != null && request.auth.token.email_verified == true
        && exists(/databases/$(database)/documents/team/$(myEmail()));
    }
    function lvl(k) { return get(/databases/$(database)/documents/team/$(myEmail())).data.p.get(k, 0); }
    function canView(k) { return isAdmin() || (isMember() && lvl(k) >= 1); }
    function canEdit(k) { return isAdmin() || (isMember() && lvl(k) >= 2); }
    match /team/{email} {
      allow get: if canEdit('team') || (request.auth != null && request.auth.token.email_verified == true && myEmail() == email);
      allow list: if canEdit('team');
      allow create, update: if canEdit('team') && email == email.lower() && email.matches('[^@ ]+@[^@ ]+')
        && request.resource.data.keys().hasOnly(['name','role','p','by','at'])
        && request.resource.data.p is map && request.resource.data.p.size() <= 30
        && request.resource.data.role in ['cofounder','manager','viewer','custom']
        && (!('name' in request.resource.data) || (request.resource.data.name is string && request.resource.data.name.size() <= 60));
      allow delete: if canEdit('team');
    }
    // What changed in the team, newest first on the page. Never edited.
    match /teamLog/{id} {
      allow read: if canEdit('team');
      allow create: if canEdit('team') && request.resource.data.keys().hasOnly(['t','by','at'])
        && request.resource.data.t is string && request.resource.data.t.size() <= 300 && request.resource.data.at == request.time;
      allow update, delete: if false;
    }
    // Team switches. teamCfg/beta.stats: every Beta tester may view HV Reset analytics (read below).
    match /teamCfg/{doc} {
      allow get: if request.auth != null;
      allow list: if false;
      allow write: if canEdit('team') && doc == 'beta' && request.resource.data.keys().hasOnly(['stats','at','by'])
        && request.resource.data.stats is bool;
    }
    function betaStats() {
      return request.auth != null
        && exists(/databases/$(database)/documents/teamCfg/beta)
        && get(/databases/$(database)/documents/teamCfg/beta).data.get('stats', false) == true
        && exists(/databases/$(database)/documents/beta/$(request.auth.uid))
        && get(/databases/$(database)/documents/beta/$(request.auth.uid)).data.get('on', false) == true;
    }
    // HV Vault: a signed-in user can read/write only their own data under users/{uid}/
    match /users/{uid}/{document=**} {
      allow read, write: if request.auth != null && request.auth.uid == uid;
    }
    // Old Harsh Reset sync (before Google accounts). Nothing writes here any more: plans now live
    // under users/{uid}/reset. Kept read-only so old data can still be copied into an account.
    // A doc can only be opened by its long private ID (no listing), and the owner's old shared
    // plan, whose ID is in the public page code, can only be opened by the admin.
    match /reset/{syncId} {
      allow get: if syncId.size() >= 32
        && (syncId != 'ffe5a55d2c34804c295003af2d6f0a7974d5fcee' || isAdmin());
      allow list, write: if false;
    }
    // HV Test scorecards: anyone can check one by its ID; created once, never changed. Only the admin can list or delete.
    match /scorecards/{id} {
      allow get: if true;
      allow list: if canView('t_scores');
      allow delete: if canEdit('t_scores');
      allow update: if false;
      allow create: if id.matches('HVT-[A-Z]{2}-[2-9A-HJ-NP-Z]{4}-[2-9A-HJ-NP-Z]{4}')
        && request.resource.data.keys().hasOnly(['v','id','name','test','testTitle','category','completedAt','score','level','answered','total','skills','strengths','focus'])
        && request.resource.data.v == 1
        && request.resource.data.id == id
        && request.resource.data.name is string && request.resource.data.name.size() > 0 && request.resource.data.name.size() <= 60
        && request.resource.data.test is string && request.resource.data.test.size() <= 60
        && request.resource.data.testTitle is string && request.resource.data.testTitle.size() <= 80
        && request.resource.data.category is string && request.resource.data.category.size() <= 60
        && request.resource.data.level is string && request.resource.data.level.size() <= 40
        && request.resource.data.completedAt is timestamp
        && request.resource.data.completedAt > request.time - duration.value(2, 'd')
        && request.resource.data.completedAt < request.time + duration.value(10, 'm')
        && request.resource.data.score is int && request.resource.data.score >= 0 && request.resource.data.score <= 100
        && request.resource.data.answered is int && request.resource.data.total is int
        && request.resource.data.answered >= 0 && request.resource.data.answered <= request.resource.data.total && request.resource.data.total <= 200
        && request.resource.data.skills is list && request.resource.data.skills.size() <= 20
        && request.resource.data.strengths is list && request.resource.data.strengths.size() <= 5
        && request.resource.data.focus is list && request.resource.data.focus.size() <= 5
        && maturityCard(request.resource.data);
    }
    // Consistency checks for the Maturity Assessment (the only test that issues scorecards). A record
    // must look like one the test really produces: 28-30 questions, all answered; the level matching the
    // score band; the 10 dimensions in order, each 0-10; an overall score in line with them (the test may
    // take up to 5 points off for answer timing); 3 strengths and 3 focus areas from those dimensions.
    // A new test that issues scorecards needs its own check here.
    function skillOk(s, name) {
      return s is map && s.keys().hasOnly(['name','score']) && s.name == name && s.score is int && s.score >= 0 && s.score <= 10;
    }
    function levelOk(score, level) {
      return (score <= 40 && level == 'Developing') || (score > 40 && score <= 65 && level == 'Emerging')
        || (score > 65 && score <= 85 && level == 'Grounded') || (score > 85 && level == 'Highly Consistent');
    }
    function maturitySkills(k, dims) {
      return k.size() == 10
        && skillOk(k[0], dims[0]) && skillOk(k[1], dims[1]) && skillOk(k[2], dims[2]) && skillOk(k[3], dims[3]) && skillOk(k[4], dims[4])
        && skillOk(k[5], dims[5]) && skillOk(k[6], dims[6]) && skillOk(k[7], dims[7]) && skillOk(k[8], dims[8]) && skillOk(k[9], dims[9]);
    }
    function maturityCard(d) {
      let dims = ['Emotional Control','Accountability','Self-Awareness','Handling Conflict','Relationships',
                  'Decision-Making','Patience','Empathy','Responsibility','Long-Term Thinking'];
      let k = d.skills;
      return d.id.matches('HVT-MA-.*') && d.test == 'maturity-assessment' && d.testTitle == 'Maturity Assessment' && d.category == 'Personal Growth'
        && d.total >= 28 && d.total <= 30 && d.answered == d.total
        && levelOk(d.score, d.level)
        && maturitySkills(k, dims)
        && sumOk(d.score, k[0].score + k[1].score + k[2].score + k[3].score + k[4].score + k[5].score + k[6].score + k[7].score + k[8].score + k[9].score)
        && d.strengths.size() == 3 && d.focus.size() == 3
        && d.strengths.hasOnly(dims) && d.focus.hasOnly(dims) && !d.strengths.hasAny(d.focus);
    }
    function sumOk(score, sum) { return score >= sum - 11 && score <= sum + 6; }
    // HV World live settings (maintenance, banner, home page switches): anyone can read, only the admin can change
    match /config/{doc} {
      allow get: if doc == 'site';
      allow list: if false;
      allow write: if canEdit('site') && doc == 'site'
        && request.resource.data.keys().hasOnly(['json', 'updatedAt', 'by'])
        && request.resource.data.json is string && request.resource.data.json.size() < 100000;
    }
    // Anonymous usage counts (HV analytics, /a.js). Only +1 counters per product per day, week, month or first-visit
    // cohort; no IDs or personal data. Anyone can add to them, only the admin can read them. Nothing can be deleted.
    match /analytics/{doc} {
      allow read: if canView('an_' + doc.split('_')[0]) || (doc.matches('reset_.*') && betaStats());
      allow create: if analyticsId(doc) && request.resource.data.size() <= 60;
      allow update: if analyticsId(doc) && request.resource.data.size() <= 900
        && request.resource.data.diff(resource.data).affectedKeys().size() <= 40;
    }
    function analyticsId(doc) {
      return doc.matches('(world|test|reset|vault)_(d|c)_20[0-9]{2}-[01][0-9]-[0-3][0-9]')
        || doc.matches('(world|test|reset|vault)_w_20[0-9]{2}-W[0-5][0-9]')
        || doc.matches('(world|test|reset|vault)_m_20[0-9]{2}-[01][0-9]');
    }
    // HV Test anonymous counter: one number per test per day (+1 when someone finishes a test). Only the admin can read it.
    match /stats/{key} {
      allow read: if canView('an_test');
      allow create: if key.matches('[a-z0-9-]{1,60}_20[0-9]{2}-[01][0-9]-[0-3][0-9]')
        && request.resource.data.keys().hasOnly(['completed']) && request.resource.data.completed == 1;
      allow update: if request.resource.data.keys().hasOnly(['completed'])
        && request.resource.data.completed == resource.data.completed + 1;
    }
    // ---------- HV Reset Beta (Beta Hub in the app, /admin/beta for the admin) ----------
    function signedIn() { return request.auth != null; }
    function strMax(v, n) { return v is string && v.size() <= n; }
    // Pass numbers are handed out in order: joining writes beta/{uid} and moves betaMeta/count up by one
    // in the same batch. Each side checks the other, so two people can never get the same number.
    function joining(n) {
      return !exists(/databases/$(database)/documents/beta/$(request.auth.uid))
        && existsAfter(/databases/$(database)/documents/beta/$(request.auth.uid))
        && getAfter(/databases/$(database)/documents/beta/$(request.auth.uid)).data.no == n;
    }
    match /betaMeta/{doc} {
      allow get: if signedIn() && doc == 'count';
      allow list, delete: if false;
      allow create: if signedIn() && doc == 'count' && request.resource.data.keys().hasOnly(['n']) && request.resource.data.n == 1 && joining(1);
      allow update: if signedIn() && doc == 'count' && request.resource.data.keys().hasOnly(['n'])
        && request.resource.data.n == resource.data.n + 1 && joining(request.resource.data.n);
    }
    // One record per tester: pass number, every join and leave (hist), what they used. The tester writes their own
    // (never their pass number or the admin's bonus points); only the admin can see everyone's.
    function testerOk(d) {
      return d.keys().hasOnly(['no','email','name','since','last','ver','dev','used','flags','pts','on','bonus','hist'])
        && (!('hist' in d) || (d.hist is list && d.hist.size() <= 300))
        && d.no is int && strMax(d.email, 120) && strMax(d.name, 60) && d.since is int && d.last is int
        && strMax(d.ver, 20) && strMax(d.dev, 80) && d.used is map && d.used.size() <= 60
        && d.flags is map && d.flags.size() <= 30 && d.pts is int && d.on is bool;
    }
    match /beta/{uid} {
      allow get: if (signedIn() && request.auth.uid == uid) || canView('b_testers');
      allow list: if canView('b_testers');
      allow create: if signedIn() && request.auth.uid == uid && testerOk(request.resource.data)
        && !('bonus' in request.resource.data)
        && getAfter(/databases/$(database)/documents/betaMeta/count).data.n == request.resource.data.no
        && (exists(/databases/$(database)/documents/betaMeta/count)
            ? get(/databases/$(database)/documents/betaMeta/count).data.n + 1 == request.resource.data.no
            : request.resource.data.no == 1);
      allow update: if (signedIn() && request.auth.uid == uid && testerOk(request.resource.data)
          && !request.resource.data.diff(resource.data).affectedKeys().hasAny(['no','bonus']))
        || (canEdit('b_testers') && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['bonus'])
          && request.resource.data.bonus is int);
      allow delete: if canEdit('b_testers');
    }
    // Leaderboard: a short name and points, readable by signed-in testers (no emails here).
    match /betaBoard/{uid} {
      allow read: if signedIn();
      allow create, update: if signedIn() && request.auth.uid == uid
        && request.resource.data.keys().hasOnly(['n','pts','no'])
        && strMax(request.resource.data.n, 30) && request.resource.data.pts is int
        && request.resource.data.pts >= 0 && request.resource.data.pts <= 1000000 && request.resource.data.no is int;
      allow delete: if (signedIn() && request.auth.uid == uid) || canEdit('b_testers');
    }
    // Bug reports: a tester sends one (with an optional screenshot) and can see only their own.
    // Only the admin sees all, groups them and changes their status.
    match /betaReports/{id} {
      allow get, list: if canView('b_reports') || (signedIn() && resource.data.uid == request.auth.uid);
      allow create: if signedIn() && request.resource.data.uid == request.auth.uid
        && request.resource.data.keys().hasOnly(['uid','kind','title','note','shot','scr','dev','ver','ctx','at','status','name','email','no'])
        && request.resource.data.kind == 'bug' && request.resource.data.status == 'new'
        && strMax(request.resource.data.title, 120) && request.resource.data.title.size() > 0
        && strMax(request.resource.data.note, 2000) && strMax(request.resource.data.shot, 450000)
        && strMax(request.resource.data.scr, 40) && strMax(request.resource.data.dev, 80) && strMax(request.resource.data.ver, 20)
        && strMax(request.resource.data.ctx, 3000) && strMax(request.resource.data.name, 60) && strMax(request.resource.data.email, 120)
        && request.resource.data.no is int && request.resource.data.at == request.time;
      allow update: if canEdit('b_reports')
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['status','group','sev','fixVer','fixedAt','reply']);
      allow delete: if canEdit('b_reports');
    }
    // Ideas: every signed-in tester can read them and vote once (a vote doc under the idea, plus the
    // count moving by exactly one in the same write). The admin moves them along the roadmap.
    function voteDoc(id) { return /databases/$(database)/documents/betaIdeas/$(id)/v/$(request.auth.uid); }
    match /betaIdeas/{id} {
      allow read: if signedIn();
      allow create: if signedIn() && request.resource.data.uid == request.auth.uid
        && request.resource.data.keys().hasOnly(['uid','name','t','note','at','votes','stage','group'])
        && strMax(request.resource.data.t, 120) && request.resource.data.t.size() > 0 && strMax(request.resource.data.note, 1000)
        && strMax(request.resource.data.name, 60) && request.resource.data.at == request.time
        && request.resource.data.votes == 0 && request.resource.data.stage == 'new' && request.resource.data.group == '';
      allow update: if canEdit('b_ideas')
        || (signedIn() && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['votes'])
          && ((request.resource.data.votes == resource.data.votes + 1 && !exists(voteDoc(id)) && existsAfter(voteDoc(id)))
           || (request.resource.data.votes == resource.data.votes - 1 && exists(voteDoc(id)) && !existsAfter(voteDoc(id)))));
      allow delete: if canEdit('b_ideas');
      match /v/{uid} {
        allow get: if signedIn() && request.auth.uid == uid;
        allow list: if false;
        allow create: if signedIn() && request.auth.uid == uid && request.resource.data.keys().hasOnly(['at'])
          && getAfter(/databases/$(database)/documents/betaIdeas/$(id)).data.votes == get(/databases/$(database)/documents/betaIdeas/$(id)).data.votes + 1;
        allow delete: if signedIn() && request.auth.uid == uid
          && getAfter(/databases/$(database)/documents/betaIdeas/$(id)).data.votes == get(/databases/$(database)/documents/betaIdeas/$(id)).data.votes - 1;
      }
    }
    // Weekly survey and quick poll answers: one per person per survey/poll, never changed afterwards.
    match /betaAnswers/{id} {
      allow get: if canView('b_survey') || (signedIn() && resource.data.uid == request.auth.uid);
      allow list: if canView('b_survey');
      allow create: if signedIn() && request.resource.data.uid == request.auth.uid
        && request.resource.data.keys().hasOnly(['uid','kind','key','a','at'])
        && request.resource.data.kind in ['survey','poll'] && request.resource.data.key is string
        && request.resource.data.key.matches('[A-Za-z0-9-]{1,40}') && id == request.auth.uid + '_' + request.resource.data.key
        && request.resource.data.a is map && request.resource.data.a.size() <= 12 && request.resource.data.at == request.time;
      allow update: if false;
      allow delete: if canEdit('b_survey');
    }
    // Private data link: a tester's own export, opened only by its long random key (no listing).
    match /betaShare/{key} {
      allow get: if key.size() >= 32;
      allow list: if false;
      allow create: if signedIn() && key.size() >= 32 && request.resource.data.uid == request.auth.uid
        && request.resource.data.keys().hasOnly(['uid','json','csv','at']) && request.resource.data.json is string && request.resource.data.csv is string
        && request.resource.data.at is int;
      allow update: if signedIn() && resource.data.uid == request.auth.uid && request.resource.data.uid == request.auth.uid
        && request.resource.data.keys().hasOnly(['uid','json','csv','at']) && request.resource.data.json is string && request.resource.data.csv is string;
      allow delete: if signedIn() && resource.data.uid == request.auth.uid;
    }
    // Beta Hub content the admin posts: what's new, known issues, roadmap notes, challenge, polls.
    match /betaConfig/{doc} {
      allow get: if signedIn() && doc == 'hub';
      allow list: if false;
      allow write: if canEdit('b_hub') && doc == 'hub'
        && request.resource.data.keys().hasOnly(['json','updatedAt']) && request.resource.data.json is string && request.resource.data.json.size() < 200000;
    }
  }
}

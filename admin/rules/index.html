/**
 * Haritt Developers · enquiry email
 * The Start a project form on developers.haritt.com saves each enquiry to Firestore (seen in /admin/enquiries)
 * and then POSTs it here. This script emails it to developers@haritt.com in the site's own design.
 *
 * Two emails go out, both FROM developers@haritt.com:
 *   1. to the team (developers@haritt.com): the full enquiry, reply-to set to the person who wrote in
 *   2. to the person who sent the form: a thank-you with a copy of what they sent
 * developers@haritt.com must be a "Send mail as" address in the Gmail that runs this script (harittdevelopers@gmail.com, where Cloudflare forwards developers@haritt.com),
 * the same way harsh@ and hello@ were set up (admin/launch, step 5). Without it, mail goes from the Gmail address.
 *
 * Set up once (script.google.com, signed in as harittdevelopers@gmail.com):
 *   New project → paste this file → Deploy → New deployment → Web app
 *   Execute as: Me · Who has access: Anyone → Deploy → copy the Web app URL into ENQ_HOOK in build.py.
 */
var TO = "developers@haritt.com", FROM = "developers@haritt.com", FROM_NAME = "Haritt Developers";
var LOGO = "https://developers.haritt.com/assets/logo-256.png";
var ADMIN = "https://hvworld.haritt.com/admin/enquiries/";

function doPost(e) {
  try {
    var d = JSON.parse((e && e.postData && e.postData.contents) || "{}");
    var clean = function (k, max) { return String(d[k] == null ? "" : d[k]).replace(/\s+$/g, "").slice(0, max); };
    var x = { id: clean("id", 40), name: clean("name", 100), email: clean("email", 200), company: clean("company", 200), site: clean("site", 200),
              services: clean("services", 400), budget: clean("budget", 60), timeline: clean("timeline", 60), desc: clean("desc", 5000) };
    if (x.name.length < 2 || !/^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(x.email) || x.desc.length < 10) return out({ ok: false });
    // at most 30 emails an hour, so a flood of fake posts can't fill the inbox
    var c = CacheService.getScriptCache(), n = Number(c.get("n") || 0);
    if (n >= 30) return out({ ok: false, busy: true });
    c.put("n", String(n + 1), 3600);
    var from = GmailApp.getAliases().indexOf(FROM) >= 0 ? { from: FROM } : {};
    GmailApp.sendEmail(TO, "New enquiry: " + (x.company || x.name) + (x.services ? " · " + x.services.split(",")[0] : ""), text(x),
      Object.assign({ htmlBody: html(x), replyTo: x.email, name: "Haritt Developers website" }, from));
    // the thank-you to the person, at most once a day per address (so the form can't be used to spam someone)
    var key = "to:" + x.email.toLowerCase();
    if (!c.get(key)) {
      c.put(key, "1", 21600);
      GmailApp.sendEmail(x.email, "We've got your enquiry · Haritt Developers", thanksText(x),
        Object.assign({ htmlBody: thanks(x), replyTo: TO, name: FROM_NAME }, from));
    }
    return out({ ok: true });
  } catch (err) { return out({ ok: false }); }
}
